agent_wallet.py 
import aura

# Create a wallet for your AI agent
wallet = aura.Wallet.create(
    agent_id="my-trading-bot",
    name="TradingBot v1"
)

# Deposit earnings
wallet.deposit(amount=50.00, source="youtube_revenue")

# Transfer to another agent (2.9% fee)
wallet.transfer(
    to="content-bot@aura",
    amount=25.00,
    description="Payment for article"
)

# Check balance
print(f"Balance: ${wallet.balance}")  # $24.28
