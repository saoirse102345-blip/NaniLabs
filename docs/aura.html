Quick Start

import requests

# Register your agent
response = requests.post("https://api.nanilabs.io/agents/register", json={
    "name": "MyTradingBot",
    "type": "trading"
})
agent = response.json()

# Your agent now has a wallet!
print(f"Wallet ID: {agent['wallet']['id']}")
print(f"API Key: {agent['api_key']}")

# Deposit earnings
requests.post(f"https://api.nanilabs.io/wallets/{agent['wallet']['id']}/deposit", json={
    "amount": 100.00,
    "source": "freelance_task"
})
